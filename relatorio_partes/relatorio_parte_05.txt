const pedidosFinalizar = async (req, res) => {
    const partes = new URL(req.url, "http://localhost:3000").pathname.split("/");
    const carrinhoId = Number(partes[2]);
    const userId = Number(req.usuario?.id);

    if (!Number.isInteger(carrinhoId) || carrinhoId <= 0) {
        return responder(res, 400, {
            mensagem: "ID do carrinho invÃ¡lido."
        });
    }

    if (!Number.isInteger(userId) || userId <= 0) {
        return responder(res, 401, {
            mensagem: "UsuÃ¡rio nÃ£o autenticado."
        });
    }

    const client = await pool.connect();

    try {
        await client.query("BEGIN");

        // Busca e bloqueia o carrinho.
        const carrinhoResult = await client.query(
            `SELECT id, user_id, status_id, oculto
             FROM carrinho
             WHERE id = $1
             FOR UPDATE`,
            [carrinhoId]
        );

        if (carrinhoResult.rows.length === 0) {
            await client.query("ROLLBACK");
            return responder(res, 404, {
                mensagem: "Carrinho nÃ£o encontrado."
            });
        }

        const carrinho = carrinhoResult.rows[0];

        if (Number(carrinho.user_id) !== userId) {
            await client.query("ROLLBACK");
            return responder(res, 403, {
                mensagem: "VocÃª nÃ£o tem permissÃ£o para finalizar este carrinho."
            });
        }

        if (carrinho.oculto) {
            await client.query("ROLLBACK");
            return responder(res, 409, {
                mensagem: "Este carrinho nÃ£o estÃ¡ disponÃ­vel."
            });
        }

        if (Number(carrinho.status_id) === 401) {
            await client.query("ROLLBACK");
            return responder(res, 409, {
                mensagem: "Este carrinho jÃ¡ foi finalizado."
            });
        }

        // Busca os produtos e bloqueia as linhas de estoque.
        const produtosResult = await client.query(
            `SELECT
                pc.produto_id,
                pc.quantidade,
                p.nome,
                p.valor,
                p.quantidade AS quantidade_estoque
             FROM produtos_carrinho pc
             INNER JOIN produtos p ON p.id = pc.produto_id
             WHERE pc.carrinho_id = $1
             ORDER BY pc.produto_id
             FOR UPDATE OF p`,
            [carrinhoId]
        );

        if (produtosResult.rows.length === 0) {
            await client.query("ROLLBACK");
            return responder(res, 400, {
                mensagem: "NÃ£o Ã© possÃ­vel finalizar um carrinho vazio."
            });
        }

        const produtosInvalidos = produtosResult.rows.filter(produto => {
            const quantidade = Number(produto.quantidade);
            const estoque = Number(produto.quantidade_estoque);

            return (
                !Number.isInteger(quantidade) ||
                quantidade <= 0 ||
                quantidade > estoque
            );
        });

        if (produtosInvalidos.length > 0) {
            await client.query("ROLLBACK");
            return responder(res, 409, {
                mensagem: "Estoque insuficiente ou quantidade invÃ¡lida.",
                produtos: produtosInvalidos.map(produto => ({
                    produto_id: produto.produto_id,
                    nome: produto.nome,
                    quantidade_solicitada: produto.quantidade,
                    quantidade_estoque: produto.quantidade_estoque
                }))
            });
        }

        // Cria o pedido como pendente (status 200).
        const pedidoResult = await client.query(
            `INSERT INTO pedidos (user_id, status_id, oculto)
             VALUES ($1, $2, false)
             RETURNING id, user_id, data_pedido, status_id, oculto`,
            [userId, 200]
        );

        const pedido = pedidoResult.rows[0];
        const produtosPedido = [];

        for (const produto of produtosResult.rows) {
            const quantidade = Number(produto.quantidade);
            const valorUnitario = Number(produto.valor);

            const produtoPedidoResult = await client.query(
                `INSERT INTO produtos_pedido
                    (pedido_id, produto_id, quantidade, valor_unitario)
                 VALUES ($1, $2, $3, $4)
                 RETURNING *`,
                [
                    pedido.id,
                    produto.produto_id,
                    quantidade,
                    valorUnitario
                ]
            );

            const estoqueResult = await client.query(
                `UPDATE produtos
                 SET quantidade = quantidade - $1
                 WHERE id = $2
                   AND quantidade >= $1
                 RETURNING id, nome, quantidade`,
                [quantidade, produto.produto_id]
            );

            if (estoqueResult.rows.length === 0) {
                throw new Error(
                    `NÃ£o foi possÃ­vel atualizar o estoque do produto ${produto.produto_id}.`
                );
            }

            produtosPedido.push({
                ...produtoPedidoResult.rows[0],
                nome: produto.nome,
                estoque_atual: estoqueResult.rows[0].quantidade
            });
        }

        // Marca o carrinho como finalizado.
        await client.query(
            `UPDATE carrinho
             SET status_id = 401
             WHERE id = $1`,
            [carrinhoId]
        );

        await client.query("COMMIT");

        return responder(res, 201, {
            mensagem: "Pedido finalizado com sucesso.",
            pedido: {
                ...pedido,
                produtos: produtosPedido
            }
        });
    } catch (erro) {
        try {
            await client.query("ROLLBACK");
        } catch (erroRollback) {
            console.error("Erro ao desfazer transaÃ§Ã£o:", erroRollback);
        }

        console.error("Erro ao finalizar pedido:", erro);

        return responder(res, 500, {
            mensagem: "Erro interno ao finalizar o pedido."
        });
    } finally {
        client.release();
    }
};

        req.on("data", (chunk) => {
            if (excedeuLimite) return;

            data += chunk.toString();

            if (data.length > 1024 * 1024) {

                reject(new Error("BODY_MUITO_GRANDE"));

        });

                reject(new Error("JSON_INVALIDO"));
            }
        });

        req.on("error", (erro) => {
            if (!excedeuLimite) reject(erro);
        });
    });

            if (

                !email.trim() ||

                senha.length < 8
            ) {

                });

                }));

                return;
            }

            if (
                telefone !== undefined &&
                telefone !== null &&
                typeof telefone !== "string"
            ) {
                res.writeHead(400, {
                    "Content-Type": "application/json; charset=utf-8"
                });

                res.end(JSON.stringify({
                    mensagem: "O telefone deve ser uma string."
                }));

                return;
            }

            const nomeNormalizado = nome.trim();
            const emailNormalizado = email.trim();
            const senhaHash = await bcrypt.hash(senha, 10);

            client = await pool.connect();
            await client.query("BEGIN");

            const usuarioExiste = await client.query(
                `SELECT id
                 FROM users
                 WHERE LOWER(email) = LOWER($1)
                 LIMIT 1`,
                [emailNormalizado]
            );

            if (usuarioExiste.rowCount > 0) {
                await client.query("ROLLBACK");

                res.writeHead(409, {
                    "Content-Type": "application/json; charset=utf-8"
                });

                res.end(JSON.stringify({
                    mensagem: "JÃ¡ existe um usuÃ¡rio com este e-mail."
                }));

                return;
            }

            const resultadoRole = await client.query(
                `SELECT id, role
                 FROM roles
                 WHERE LOWER(role) = 'client'
                 LIMIT 1`
            );

            if (resultadoRole.rowCount === 0) {
                await client.query("ROLLBACK");

                res.writeHead(500, {

                });

                res.end(JSON.stringify({
                    mensagem: "A role client nÃ£o estÃ¡ cadastrada no banco."
                }));

                return;
            }

            const roleClient = resultadoRole.rows[0];

            const resultadoUsuario = await client.query(
                `INSERT INTO users
                    (nome, email, telefone, senha_hash, ativo)
                 VALUES ($1, $2, $3, $4, TRUE)
                 RETURNING id, nome, email, telefone, ativo`,
                [
                    nomeNormalizado,
                    emailNormalizado,
                    telefone || null,
                    senhaHash
                ]
            );

            const usuario = resultadoUsuario.rows[0];

            await client.query(
                `INSERT INTO users_roles (user_id, role_id)
                 VALUES ($1, $2)`,
                [usuario.id, roleClient.id]
            );

            }
        }

const usersLogin = async (req, res) => {
    let body = "";

    req.on("data", (chunk) => {
        body += chunk.toString();

        if (body.length > 1024 * 1024) {
            responder(res, 413, {
                mensagem: "Corpo da requisiÃ§Ã£o muito grande."
            });
            req.destroy();
        }
    });

    req.on("end", async () => {
        if (res.writableEnded) return;

        try {
            const dados = body ? JSON.parse(body) : {};
            const email = dados.email;
            const senha = dados.senha;

            if (
                typeof email !== "string" ||
                typeof senha !== "string" ||
                !email.trim() ||
                !senha
            ) {
                responder(res, 400, {
                    mensagem: "Informe o e-mail e a senha."
                });
                return;
            }

            const resultado =
                await userRepository.buscarPorEmailParaLogin(
                    email.trim()
                );

            if (resultado.rows.length === 0) {
                responder(res, 401, {
                    mensagem: "E-mail ou senha invÃ¡lidos."
                });
                return;
            }

            const usuario = resultado.rows[0];

            if (usuario.ativo !== true || !usuario.senha_hash) {
                responder(res, 401, {
                    mensagem: "E-mail ou senha invÃ¡lidos."
                });
                return;
            }

            const senhaCorreta = await bcrypt.compare(
                senha,
                usuario.senha_hash
            );

            if (!senhaCorreta) {
                responder(res, 401, {
                    mensagem: "E-mail ou senha invÃ¡lidos."
                });
                return;
            }

            // Retorna somente o ID de cada papel.
            const resultadoRoles =
                await roleRepository.lerPorUser(usuario.id);

            const roles = resultadoRoles.rows.map((role) => ({
                id: role.id
            }));

            const segredo = process.env.JWT_SECRET;

            if (!segredo) {
                throw new Error("JWT_SECRET nÃ£o configurado.");
            }

            // O ID continua no token para autenticaÃ§Ã£o interna,
            // mas nÃ£o serÃ¡ incluÃ­do no objeto usuario da resposta.
            const token = jwt.sign(
                {
                    sub: String(usuario.id)
                },
                segredo,
                {
                    expiresIn: "2h"
                }
            );

            // MantÃ©m o ID disponÃ­vel internamente para os logs.
            req.usuario = {
                id: usuario.id
            };

            req.logMensagem = "Login realizado com sucesso.";

            responder(res, 200, {
                mensagem: "Login realizado com sucesso.",
                token,
                usuario: {
                    nome: usuario.nome,
                    email: usuario.email
                },
                roles
            });

const jwt = require("jsonwebtoken");

const autenticar = (req, res, proximo) => {
    const responder = (status, dados) => {
        res.writeHead(status, {
            "Content-Type": "application/json; charset=utf-8"
        });
        res.end(JSON.stringify(dados));
    };

    const autorizacao = req.headers.authorization;

    if (!autorizacao || !autorizacao.startsWith("Bearer ")) {
        responder(401, {
            mensagem: "Token de autenticaÃ§Ã£o nÃ£o informado."
        });
        return;
    }

    const token = autorizacao.slice(7);
    const segredo = process.env.JWT_SECRET;

    if (!segredo) {
        console.error("JWT_SECRET nÃ£o estÃ¡ configurado.");
        responder(500, {
            mensagem: "Erro de configuraÃ§Ã£o da autenticaÃ§Ã£o."
        });
        return;
    }

    try {
        const payload = jwt.verify(token, segredo);
        const userId = Number(payload.sub);

        if (!Number.isInteger(userId) || userId <= 0) {
            responder(401, {
                mensagem: "Token invÃ¡lido."
            });
            return;
        }

        req.usuario = { id: userId };
        proximo();
    } catch (erro) {
        responder(401, {
            mensagem: "Token invÃ¡lido ou expirado."
        });
    }
};

module.exports = autenticar;

const permissionRepository = require("../repositories/permissionRepository");

function normalizarRota(pathname) {
    return pathname
        .split("/")
        .map((parte) => /^\d+$/.test(parte) ? ":id" : parte)
        .join("/");
}

function responder(res, status, mensagem) {
    if (res.writableEnded) {
        return;
    }

    res.writeHead(status, {
        "Content-Type": "application/json; charset=utf-8"
    });

    res.end(JSON.stringify({ mensagem }));
}

async function autorizar(req, res, proximo) {
    try {
        if (!req.usuario || !req.usuario.id) {
            responder(res, 401, "UsuÃ¡rio nÃ£o autenticado.");
            return;
        }

        const url = new URL(req.url, "http://localhost");
        const rota = normalizarRota(url.pathname);
        const metodo = req.method.toUpperCase();

        // A consulta individual de pedido possui regras prÃ³prias
        // no controlador pedidosConsultar.js.
        // A autenticaÃ§Ã£o JWT jÃ¡ foi validada antes deste middleware.
        if (
            metodo === "GET" &&
            /^\/pedido\/\d+$/.test(url.pathname)
        ) {
            proximo();
            return;
        }

        // Verifica se a rota e o mÃ©todo estÃ£o cadastrados.
        const permissao =
            await permissionRepository.buscarPermissaoPorRotaEMetodo(
                rota,
                metodo
            );

        if (!permissao) {
            responder(
                res,
                403,
                "Esta operaÃ§Ã£o nÃ£o possui permissÃ£o cadastrada."
            );
            return;
        }

        // Verifica se alguma role do usuÃ¡rio concede essa permissÃ£o.
        const permitido =
            await permissionRepository.usuarioTemPermissao(
                req.usuario.id,
                rota,
                metodo
            );

        if (!permitido) {
            responder(
                res,
                403,
                "VocÃª nÃ£o tem permissÃ£o para esta operaÃ§Ã£o."
            );
            return;
        }

        proximo();
    } catch (erro) {
        console.error("[ERRO NA AUTORIZAÃ‡ÃƒO]", erro);

        responder(
            res,
            500,
            "Erro ao verificar as permissÃµes."
        );
    }
}

module.exports = autorizar;
