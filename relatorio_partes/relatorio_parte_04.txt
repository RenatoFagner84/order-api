const caixaAtualizar = async (req, res) => {
  const id = req.url.split("/")[3];

  let body = "";

  req.on("data", (chunk) => {
    body += chunk.toString();
  });

  req.on("end", async () => {
    const dados = JSON.parse(body);

    const {
      user_id,
      pedido_id,
      tipo_movimentacao_id,
      subtotal,
      desconto,
      frete,
      acrescimo,
      total,
      forma_pagamento_id,
      pagamento_status_id,
      descricao
    } = dados;

    const caixaExiste = await caixaRepository.ler(id);

    if (caixaExiste.rows.length === 0) {
      res.writeHead(404, {
        "Content-Type": "application/json"
      });

      res.end(JSON.stringify({
        mensagem: "MovimentaÃ§Ã£o do caixa nÃ£o encontrada"
      }));

      return;
    }

    const resultado = await caixaRepository.atualizar(
      id,
      user_id,
      pedido_id,
      tipo_movimentacao_id,
      subtotal,
      desconto,
      frete,
      acrescimo,
      total,
      forma_pagamento_id,
      pagamento_status_id,
      descricao
    );

    res.writeHead(200, {
      "Content-Type": "application/json"
    });

    res.end(JSON.stringify(resultado.rows[0]));
  });
};

const cartAdicionarProduto = async (req, res) => {
    try {
        const userId = Number(req.usuario?.id);

        if (!Number.isSafeInteger(userId) || userId <= 0) {
            responder(res, 401, {
                mensagem: "UsuÃ¡rio nÃ£o autenticado."
            });
            return;
        }

        const pathname = new URL(
            req.url,
            "http://localhost"
        ).pathname;

        const carrinhoId = Number(pathname.split("/")[2]);

        if (!Number.isSafeInteger(carrinhoId) || carrinhoId <= 0) {
            responder(res, 400, {
                mensagem: "ID do carrinho invÃ¡lido."
            });
            return;
        }

        let body = "";

        for await (const chunk of req) {
            body += chunk.toString();

            if (body.length > 1024 * 1024) {
                responder(res, 413, {
                    mensagem: "Corpo da requisiÃ§Ã£o muito grande."
                });
                return;
            }
        }

        let dados;

        try {
            dados = JSON.parse(body || "{}");
        } catch {
            responder(res, 400, {
                mensagem: "Corpo da requisiÃ§Ã£o JSON invÃ¡lido."
            });
            return;
        }

        const produtoId = Number(dados.produto_id);
        const quantidade = Number(dados.quantidade);

        if (
            !Number.isSafeInteger(produtoId) ||
            produtoId <= 0 ||
            !Number.isSafeInteger(quantidade) ||
            quantidade <= 0
        ) {
            responder(res, 400, {
                mensagem:
                    "Informe produto_id e quantidade vÃ¡lidos."
            });
            return;
        }

        const carrinhoResult =
            await cartRepository.consultar(carrinhoId);

        const carrinho = carrinhoResult.rows[0];

        if (!carrinho) {
            responder(res, 404, {
                mensagem: "Carrinho nÃ£o encontrado."
            });
            return;
        }

        if (Number(carrinho.user_id) !== userId) {
            responder(res, 403, {
                mensagem:
                    "VocÃª nÃ£o tem permissÃ£o para alterar este carrinho."
            });
            return;
        }

        if (
            carrinho.oculto === true ||
            String(carrinho.status || "").toLowerCase() !== "aberto"
        ) {
            responder(res, 409, {
                mensagem:
                    "Este carrinho nÃ£o estÃ¡ disponÃ­vel para alteraÃ§Ãµes."
            });
            return;
        }

        const itemAtual = await cartRepository.consultarItem(
            carrinhoId,
            produtoId
        );

        const quantidadeFinal =
            quantidade +
            (Number(itemAtual.rows[0]?.quantidade) || 0);

        const estoque = await cartRepository.verificarEstoque(
            produtoId,
            quantidadeFinal
        );

        if (estoque.rows.length === 0) {
            responder(res, 409, {
                mensagem:
                    "Produto inexistente ou estoque insuficiente para a quantidade solicitada."
            });
            return;
        }

        const resultado = await cartRepository.adicionarProduto(
            carrinhoId,
            produtoId,
            quantidade
        );

        responder(res, 201, resultado.rows[0]);
    } catch (erro) {
        console.error(
            "[ERRO AO ADICIONAR PRODUTO AO CARRINHO]",
            erro
        );

        responder(res, 500, {
            mensagem:
                "Erro interno ao adicionar produto ao carrinho."
        });

            return;
