// Verifica se o usuÃ¡rio tem permissÃ£o para acessar uma rota
async function usuarioTemPermissao(userId, url, metodo) {
    const sql = `
        SELECT EXISTS (
            SELECT 1
            FROM users u
            INNER JOIN users_roles ur
                ON ur.user_id = u.id
            INNER JOIN roles_permissions rp
                ON rp.role_id = ur.role_id
            INNER JOIN permissions p
                ON p.id = rp.permission_id
            WHERE u.id = $1
              AND u.ativo = TRUE
              AND p.url = $2
              AND UPPER(p.metodo) = UPPER($3)
        ) AS permitido
    `;

    const resultado = await pool.query(sql, [
        userId,
        url,
        metodo
    ]);

    return resultado.rows[0]?.permitido === true;
}

            ON rp.role_id = ur.role_id

            ON p.id = rp.permission_id

const buscarPorNome = async (nome) => {
    const palavras = String(nome || "")
        .trim()
        .split(/\s+/)
        .filter(Boolean);

    if (palavras.length === 0) {
        return {
            rows: [],
            rowCount: 0
        };
    }

    const busca = palavras
        .map((palavra) =>
            `${palavra.replace(/[^\p{L}\p{N}_]/gu, "")}:*`
        )
        .filter((palavra) => palavra !== ":*")
        .join(" & ");

    if (!busca) {
        return {
            rows: [],
            rowCount: 0
        };
    }

    return await pool.query(
        `SELECT
            id,
            nome,
            email,
            telefone,
            status_id,
            ativo
         FROM users
         WHERE ativo = TRUE
           AND to_tsvector('simple', nome)
               @@ to_tsquery('simple', $1)
         ORDER BY nome`,
        [busca]
    );
};

    // Cadastro pÃºblico de cliente
    if (req.method === "POST" && caminho === "/users/inserir") {
        await user.usersInserir(req, res);
        return;
    }

    // Listar usuÃ¡rios
    if (req.method === "GET" && caminho === "/users/ler") {
        await user.usersLer(req, res);
        return;
    }

    // Consultar usuÃ¡rio por ID
    if (
        req.method === "GET" &&
        /^\/users\/ler\/\d+$/.test(caminho)
    ) {
        await user.usersLerId(req, res);
        return;
    }

    // Buscar usuÃ¡rios por nome
    if (req.method === "GET" && caminho === "/users/buscar") {
        await user.usersBuscar(req, res);
        return;
    }

    // Consultar histÃ³rico de status de um usuÃ¡rio especÃ­fico
    if (
        req.method === "GET" &&
        /^\/users\/historico-status\/\d+$/.test(caminho)
    ) {
        await user.usersHistoricoStatus(req, res);
        return;
    }

    // Atualizar usuÃ¡rio por ID
    if (
        req.method === "PUT" &&
        /^\/users\/atualizar\/\d+$/.test(caminho)
    ) {
        await user.usersAtualizar(req, res);
        return;
    }

    // Desativar usuÃ¡rio por ID
    if (
        req.method === "DELETE" &&
        /^\/users\/deletar\/\d+$/.test(caminho)
    ) {
        await user.usersDeletar(req, res);
        return;
    }

    // Identifica rotas antigas ou caminhos invÃ¡lidos
    if (
        caminho === "/users/historico-status" ||
        caminho === "/users/atualizar" ||
        caminho === "/users/deletar" ||
        caminho === "/users/inserir-interno"
    ) {
        responder(
            res,
            404,
            "Esta rota nÃ£o existe. Confira o caminho e informe o ID quando necessÃ¡rio."
        );
        return;
    }

    responder(res, 404, "Rota de usuÃ¡rios nÃ£o encontrada.");
};
